{{- $url := urls.Parse .Destination -}} {{- $src := $url.String -}} {{- $resource := false -}} {{- if not $url.IsAbs -}} {{- $path := strings.TrimPrefix "./" $url.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $resource = . -}} {{- $src = .RelPermalink -}} {{- with $url.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $url.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- $processableFormats := slice "jpg" "jpeg" "png" "webp" -}} {{- $isProcessable := and $resource (in $processableFormats $resource.MediaType.SubType) -}} {{- if $isProcessable -}} {{- $sizes := slice 480 800 1200 1600 -}} {{- else -}} {{- end -}}
